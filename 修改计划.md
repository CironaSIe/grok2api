## 16. Python daemon 子进程 SSO2OAUTH 转换

> **触发：** Go 纯 `tls-client` 无法过 Cloudflare `accounts.x.ai` JS challenge；`curl_cffi` `impersonate=chrome136` 实测可过（`~/tmp/opencode/sso2oauth_result.json` 已验证）。
> **根因：** Go 生态无 curl_cffi 等价物（`tls-client` 缺 HTTP/2 SETTINGS 帧和 BoringSSL 精度），FlareSolverr 在 Termux 不可用（无 Chromium）。
> **方案：** Python 常驻 daemon（curl_cffi + 本地 HTTP）接管 SSO2OAUTH 全流程（Phase 00-05g）；Go 端作为 HTTP 客户端调用；Python 不可用时直接失败。
> **质量标尺：** 完整、通用、明确、无副作用。

### 16.1 现状与缺陷

| # | 位置 | 缺陷 | 影响 |
|---|------|------|------|
| D1 | `sso_build.go:83-85` | `ssoBuildFlow` cookies 只有 `sso`/`sso-rw`，`lease.CFCookies` 未注入（§15 追加修复已缓解但治标） | CF HTML 步骤 403 |
| D2 | `tlsclient.go:58-72` | `bogdanfinn/tls-client` 用 uTLS（Go crypto/tls fork）而非 BoringSSL；v1.15.1 无 Chrome_136 profile（只有 Chrome_144 近似）；TLS 扩展字节序/GREASE 生成方式与 BoringSSL 有微妙差异 | CF 检测到指纹差异 → 403；回退可用但不可靠 |
| D3 | `buildclient.go:19` | ScopeBuild 用标准 `net/http`，TLS 指纹是 Go 特征 | CF 对 Build API 也可能拦截 |
| D4 | 整体架构 | 无子进程/外部进程模式，Go 单进程承担所有 HTTP 交互 | 无法利用 Python curl_cffi 的高保真指纹 |

> **复用不动：** `ClassifyConversionError`（6 类）、`conversionHTTPError`、`sso2oauth[phase]` 错误前缀格式、`xaiauth.IdentityFromTokens`、`xaiauth.ClassifyConvertBot`、`xaiauth.StableAgentIDFromSSO` — Go 端保留，不重写。

### 16.2 方案选定

| 维度 | 决策 | 依据 |
|------|------|------|
| 生命周期 | 常驻 daemon | 避免 300ms/次 Python 启动开销 |
| IPC | 本地 HTTP 服务（仅绑定 127.0.0.1） | 标准协议、并发天然、无鉴权 |
| 职责边界 | Python 完整 9 步（00-05g）+ 返回完整 phase trace | Go 可从 trace 对比自身实现缺陷 |
| 依赖分发 | `pip install curl_cffi` | 生产部署文档要求 |
| 不可用处理 | 回退 Go tls-client + 发出警告 | 回退比直接失败更友好；但 Go impersonate 精度不足（uTLS vs BoringSSL），需告警运维和前端 |
| 启动责任 | Go 自动拉起 | 运维无感 |
| 错误分类 | Go 用现有 `ClassifyConversionError` 重新分类 | 复用现有 6 类 |
| enrichment | Python 走完 Phase 05 全部 | 返回 tokens + identity + trace |

**新增 `convertViaDaemon` 方法：**

```go
func (a *Adapter) convertViaDaemon(ctx context.Context, credential accountdomain.Credential, token string, lease *infraegress.Lease) (provider.CredentialSeed, error) {
    cfg := a.config()
    cliVersion := strings.TrimSpace(cfg.BuildClientVersion)
    if cliVersion == "" { cliVersion = xaiauth.DefaultCLIVersion }
    req := sso2oauth.ConvertRequest{
        SsoToken: token,
        ProxyURL: lease.ProxyURL,
        UserAgent: lease.UserAgent,
        CFCookies: lease.CFCookies,
        CLIVersion: cliVersion,
        Options: sso2oauth.ConvertOptions{
            SoftPreflight: cfg.ConvertSoftPreflight,
            SkipInitUser:  cfg.SkipConvertInitUser,
            SkipBotReject: cfg.SkipConvertBotReject,
        },
    }
    resp, err := a.sso2oauthClient.Convert(ctx, req)
    if err != nil {
        // daemon 不可达 → network_retry
        return provider.CredentialSeed{}, fmt.Errorf("sso2oauth daemon 不可达: %w", err)
    }
    if !resp.OK {
        // Python 返回失败 → 构造 sso2oauth[phase] 格式 error 供 ClassifyConversionError 分类
        return provider.CredentialSeed{}, a.classifyDaemonError(resp)
    }
    // 成功 → Go 端用 xaiauth 解析 tokens 构造 seed
    return a.buildSeedFromDaemon(resp, credential)
}
```

**`classifyDaemonError`：** 按 `resp.ErrorStatus` 构造 wrapped error：
- 401 + phase 含 `probe_accounts` → `provider.ErrUnauthorized`
- 400 + message 含 `access_denied` → `provider.ErrAuthorizationDenied`
- 其他 → `conversionHTTPError{status: resp.ErrorStatus}`
- 前缀：`fmt.Errorf("sso2oauth[%s] %s: %w", resp.ErrorPhase, resp.ErrorMessage, wrapped)`

**`buildSeedFromDaemon`：**
- `xaiauth.IdentityFromTokens(resp.Tokens.AccessToken, resp.Tokens.IDToken)` → userID/email/teamID（复用现有，不信任 Python 的 identity 字段，Go 端独立解析 JWT）
- `xaiauth.ClassifyConvertBot(accessClaims)` → botFlag（复用现有）
- bot contaminated 且 `!cfg.SkipConvertBotReject` → `ErrBuildTokenBotContaminated`
- trace 记录到 `a.log().Debug("sso2oauth_daemon_trace", "phases", resp.Phases)`
- 构造 `provider.CredentialSeed{Provider: ProviderBuild, AuthType: AuthTypeOAuth, ...}`（与现有 line 217-222 逻辑一致）

```python
            k, _, v = kv.partition("=")
            s.cookies.set(k.strip(), v.strip(), domain=".x.ai", path="/")

    IMP = "chrome136"

    try:
        # Phase 00: probe accounts.x.ai
        r = s.get(f"{ACCOUNTS_BASE}/", impersonate=IMP, timeout=30, allow_redirects=True)
        phases.append(trace("00-probe", "GET", r))
        if r.status_code >= 400:
            return fail("probe_accounts", r, phases)

        # Phase 01: CLI stable version (soft preflight)
        if opts.get("soft_preflight"):
            r = s.get(CLI_STABLE_URL, impersonate=IMP, timeout=15)
            phases.append(trace("01-cli-version", "GET", r))

        # Phase 02: login-config (soft preflight)

            r = s.get(LOGIN_CONFIG_URL, impersonate=IMP, timeout=15,
                      headers=cli_auth_headers(cli_ver, agent_id))
            phases.append(trace("02-login-config", "GET", r))

        r = s.get(f"{AUTH_BASE}/.well-known/openid-configuration", impersonate=IMP, timeout=15)
        phases.append(trace("03-oidc-discovery", "GET", r))
        if r.status_code >= 400:
            return fail("oidc_discovery", r, phases)

        # Phase 04: device/code
        r = s.post(DEVICE_CODE_URL, impersonate=IMP, timeout=15,
                   data={"client_id": CLIENT_ID, "scope": DEFAULT_SCOPE, "referrer": "grok-build"},
                   headers=cli_form_headers(cli_ver, surface="ui"))
        phases.append(trace("04-device-code", "POST", r))
        if r.status_code >= 400:
            return fail("device_code", r, phases)
        dev = r.json()
        uc = dev["user_code"]; dc = dev["device_code"]; vc = dev["verification_uri_complete"]
        interval = dev.get("interval", 5)

        # Phase 04a: verify page
        r = s.get(vc, impersonate=IMP, timeout=30, allow_redirects=True,
                  headers=browser_html_headers(ua, "verify_page", uc))
        phases.append(trace("04a-verify-page", "GET", r))

            return fail("verify_page", r, phases)

        # Phase 04b: POST verify
        r = s.post(VERIFY_URL, impersonate=IMP, timeout=30, allow_redirects=False,
                   data={"user_code": uc},
                   headers=browser_html_headers(ua, "verify", uc))
        phases.append(trace("04b-device-verify", "POST", r))
        if r.status_code not in (302, 303):
            return fail("device_verify", r, phases)
        consent_url = r.headers.get("Location", "")

        # Phase 04b-2: GET consent page
        if consent_url:
            r = s.get(consent_url, impersonate=IMP, timeout=30, allow_redirects=True,
                      headers=browser_html_headers(ua, "consent", uc))
            phases.append(trace("04b-consent-page", "GET", r))

        # Phase 04c: POST approve
        r = s.post(APPROVE_URL, impersonate=IMP, timeout=30, allow_redirects=False,
                   data={"user_code": uc, "action": "allow", "principal_type": "User", "principal_id": ""},
                   headers=browser_html_headers(ua, "approve", uc, consent_url))
        phases.append(trace("04c-device-approve", "POST", r))
        if r.status_code not in (302, 303):
            return fail("device_approve", r, phases)

        # Phase 05: token poll
        for i in range(15):
            time.sleep(interval)
            r = s.post(TOKEN_URL, impersonate=IMP, timeout=15,
                       data={"grant_type": "urn:ietf:params:oauth:grant-type:device_code",
                             "device_code": dc, "client_id": CLIENT_ID},
                       headers=cli_form_headers(cli_ver))

                phases.append(trace("05-token-poll", "POST", r))
                return fail("token_poll", r, phases)
        return fail("token_poll_timeout", None, phases)
```

```
  │

  └─ supervisor.Start(ctx)
       ├─ exec.Command(pythonPath, scriptPath, "--port", "0")
       ├─ cmd.Env = os.Environ() + cfg.Env
       ├─ cmd.Stdout pipe → goroutine 读 "READY port=XXXX"

       ├─ NewClient("http://127.0.0.1:"+port)

            ├─ 失败 3 次 → kill + 重启

       └─ 超时 → SIGKILL
```

### 16.8 错误分类映射

Python 返回 `error_status` + `error_phase` + `error_message`。Go 端 `classifyDaemonError` 构造 wrapped error：

| Python 返回 | Go 构造 | ClassifyConversionError |
|-------------|---------|------------------------|
| status=401, phase 含 `probe_accounts` | `provider.ErrUnauthorized` | `sso_dead` |
| status=429 | `conversionHTTPError{429}` | `rate_limited` |
| status=502/503/504 | `conversionHTTPError{502}` | `network_retry` |
| status=400, msg 含 `access_denied` | `provider.ErrAuthorizationDenied` | `permanent` |
| status=400, bot_flag contaminated | `ErrBuildTokenBotContaminated` | `bot_contaminated` |
| daemon 不可达 → 回退 Go | 回退 `ssoBuildFlow.convert`；`Warn` 日志 `sso2oauth_daemon_unavailable_fallback` | Go flow 自行分类（CF 403 → `permanent` 或 `network_retry`） |
| 其他 | `conversionHTTPError{status}` | 按 status |

> error 格式：`fmt.Errorf("sso2oauth[%s] %s: %w", phase, message, wrapped)`，与现有 `sso_build.go` 一致。

```go
		},
		logger: logger,
	}
}

func (c *Client) Convert(ctx context.Context, req ConvertRequest) (*ConvertResponse, error) {
	body, err := json.Marshal(req)
	if err != nil {
		return nil, fmt.Errorf("编码请求: %w", err)
	}
	httpReq, err := http.NewRequestWithContext(ctx, http.MethodPost, c.baseURL+"/convert", bytes.NewReader(body))
	if err != nil {
		return nil, fmt.Errorf("创建请求: %w", err)
	}
	httpReq.Header.Set("Content-Type", "application/json")
	resp, err := c.httpClient.Do(httpReq)
	if err != nil {
		return nil, fmt.Errorf("sso2oauth daemon 不可达: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("sso2oauth daemon HTTP %d", resp.StatusCode)
	}
	var result ConvertResponse
	if err := json.NewDecoder(io.LimitReader(resp.Body, 1<<20)).Decode(&result); err != nil {
		return nil, fmt.Errorf("解析 daemon 响应: %w", err)
	}
	return &result, nil
}

	}
```

```go
package sso2oauth

import (
	"bufio"
	"context"
	"errors"
	"fmt"
	"io"
	"log/slog"
	"os"
	"os/exec"
	"strconv"
	"strings"
	"sync"
	"time"
)

type SupervisorConfig struct {
	Enabled         bool
	PythonPath      string
	ScriptPath      string
	Env             map[string]string
	StartupTimeout  time.Duration
	HealthInterval  time.Duration
	MaxRestarts     int
	RestartWindow   time.Duration
}

type Supervisor struct {
	cfg           SupervisorConfig
	client        *Client
	cmd           *exec.Cmd
	port          int
	mu            sync.Mutex
	stop          chan struct{}
	wg            sync.WaitGroup
	restartCount  int
	restartWindow time.Time
	logger        *slog.Logger
}

func NewSupervisor(cfg SupervisorConfig, logger *slog.Logger) *Supervisor {
	if logger == nil {
		logger = slog.Default()
	}
	if cfg.StartupTimeout <= 0 {
		cfg.StartupTimeout = 15 * time.Second
	}
	if cfg.HealthInterval <= 0 {
		cfg.HealthInterval = 30 * time.Second
	}
	if cfg.MaxRestarts <= 0 {
		cfg.MaxRestarts = 5
	}
	if cfg.RestartWindow <= 0 {
		cfg.RestartWindow = 10 * time.Minute
	}
	return &Supervisor{cfg: cfg, logger: logger, stop: make(chan struct{})}
}

func (s *Supervisor) Start(ctx context.Context) error {
	if err := s.startProcess(ctx); err != nil {
		return err
	}
	s.wg.Add(1)
	go s.healthLoop()
	return nil
}

func (s *Supervisor) startProcess(ctx context.Context) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	cmd := exec.CommandContext(ctx, s.cfg.PythonPath, s.cfg.ScriptPath, "--port", "0")
	cmd.Env = os.Environ()
	for k, v := range s.cfg.Env {
		cmd.Env = append(cmd.Env, k+"="+v)
	}
	// 环境变量传端口占位（Python 端 --port 0 让 OS 分配，但 Env 可覆盖）
	stdout, err := cmd.StdoutPipe()
	if err != nil {
		return fmt.Errorf("创建 stdout pipe: %w", err)
	}
	if err := cmd.Start(); err != nil {
		return fmt.Errorf("启动 Python daemon: %w", err)
	}
	s.cmd = cmd
	// 等待 READY 信号
	readyCh := make(chan int, 1)
	go func() {
		scanner := bufio.NewScanner(stdout)
		for scanner.Scan() {
			line := strings.TrimSpace(scanner.Text())
			if strings.HasPrefix(line, "READY port=") {
				portStr := strings.TrimPrefix(line, "READY port=")
				if port, err := strconv.Atoi(portStr); err == nil {
					readyCh <- port
					// 继续读 stdout（避免 pipe 阻塞），丢弃后续输出
					go io.Copy(io.Discard, stdout)
					return
				}
			}
		}
		readyCh <- 0
	}()
	select {
	case port := <-readyCh:
		if port == 0 {
			cmd.Process.Kill()
			return errors.New("Python daemon 未输出有效 READY 信号")
		}
		s.port = port
		s.client = NewClient(fmt.Sprintf("http://127.0.0.1:%d", port), s.logger)
	case <-time.After(s.cfg.StartupTimeout):
		cmd.Process.Kill()
		return fmt.Errorf("等待 Python daemon READY 超时 (%s)", s.cfg.StartupTimeout)
	}
	return nil
}

func (s *Supervisor) Stop() error {
	close(s.stop)
	s.wg.Wait()
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.cmd == nil || s.cmd.Process == nil {
		return nil
	}
	// 优雅关闭
	if s.client != nil {
		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		req, _ := http.NewRequestWithContext(ctx, "GET", s.client.baseURL+"/shutdown", nil)
		resp, err := s.client.httpClient.Do(req)
		if err == nil {
			resp.Body.Close()
		}
	}
	done := make(chan struct{})
	go func() { s.cmd.Wait(); close(done) }()
	select {
	case <-done:
	case <-time.After(5 * time.Second):
		s.cmd.Process.Kill()
	}
	return nil
}

func (s *Supervisor) Client() *Client {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.client
}

func (s *Supervisor) IsRunning() bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.cmd != nil && s.cmd.Process != nil
}

func (s *Supervisor) healthLoop() {
	defer s.wg.Done()
	ticker := time.NewTicker(s.cfg.HealthInterval)
	defer ticker.Stop()
	failures := 0
	for {
		select {
		case <-s.stop:
			return
		case <-ticker.C:
			if s.client == nil {
				continue
			}
			ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
			err := s.client.Health(ctx)
			cancel()
			if err != nil {
				failures++
				if failures >= 3 {
					s.logger.Warn("sso2oauth_daemon_health_failed", "failures", failures, "error", err)
					s.restart()
					failures = 0
				}
			} else {
				failures = 0
			}
		}
	}
}

func (s *Supervisor) restart() {
	s.mu.Lock()
	now := time.Now()
	if now.Sub(s.restartWindow) > s.cfg.RestartWindow {
		s.restartCount = 0
		s.restartWindow = now
	}
	s.restartCount++
	if s.restartCount > s.cfg.MaxRestarts {
		s.logger.Error("sso2oauth_daemon_restart_limit_exceeded", "count", s.restartCount, "window", s.cfg.RestartWindow)
		s.mu.Unlock()
		return
	}
	if s.cmd != nil && s.cmd.Process != nil {
		s.cmd.Process.Kill()
	}
	s.mu.Unlock()
	time.Sleep(5 * time.Second)
	ctx, cancel := context.WithTimeout(context.Background(), s.cfg.StartupTimeout)
	defer cancel()
	if err := s.startProcess(ctx); err != nil {
		s.logger.Error("sso2oauth_daemon_restart_failed", "error", err)
	}
}
```

```python
#!/usr/bin/env python3
"""SSO2OAUTH daemon — curl_cffi powered local HTTP service."""
import json, sys, os, signal, threading, argparse
from http.server import ThreadingHTTPServer, BaseHTTPRequestHandler

try:
    from curl_cffi import requests as cffi_requests
    CFFI_VERSION = getattr(cffi_requests, '__version__', 'unknown')
except ImportError:
    CFFI_VERSION = 'not-installed'

class DaemonHandler(BaseHTTPRequestHandler):
    def do_POST(self):
        if self.path == "/convert":
            self._handle_convert()
        else:
            self._json_response(404, {"error": "not found"})

    def do_GET(self):
        if self.path == "/health":
            self._json_response(200, {"status": "ok", "curl_cffi_version": CFFI_VERSION})
        elif self.path == "/shutdown":
            self._json_response(200, {"status": "shutting_down"})
            threading.Timer(1.0, lambda: os._exit(0)).start()
        else:
            self._json_response(404, {"error": "not found"})

    def _handle_convert(self):
        length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(length) if length > 0 else b"{}"
        try:
            req = json.loads(body)
        except json.JSONDecodeError as e:
            self._json_response(400, {"ok": False, "error_phase": "invalid_request", "error_message": str(e)})
            return
        from sso2oauth_flow import convert
        result = convert(req)
        self._json_response(200, result)

    def _json_response(self, status, obj):
        data = json.dumps(obj).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def log_message(self, *args): pass  # 静默

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--port", type=int, default=0)
    args = parser.parse_args()
    server = ThreadingHTTPServer(("127.0.0.1", args.port), DaemonHandler)
    actual_port = server.server_address[1]
    print(f"READY port={actual_port}", flush=True)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass

if __name__ == "__main__":
    main()
```

```python
    try:
        # Phase 00: probe accounts.x.ai
        r = s.get(f"{ACCOUNTS_BASE}/", impersonate=IMP, timeout=30, allow_redirects=True)
        phases.append(trace("00-probe", "GET", r))
        if r.status_code >= 400:
            return _fail("probe_accounts", r, phases)

        # Phase 01: CLI stable version (soft preflight)
        if opts.get("soft_preflight"):
            r = s.get(CLI_STABLE_URL, impersonate=IMP, timeout=15,
                      headers={"Accept": "*/*", "Accept-Encoding": "gzip, br, deflate"})
            phases.append(trace("01-cli-version", "GET", r))

        # Phase 02: login-config (soft preflight)
        if opts.get("soft_preflight"):
            r = s.get(LOGIN_CONFIG_URL, impersonate=IMP, timeout=15,
                      headers=login_config_headers(cli_ver, agent_id))
            phases.append(trace("02-login-config", "GET", r))

        # Phase 03: OIDC discovery
        r = s.get(f"{AUTH_BASE}/.well-known/openid-configuration", impersonate=IMP, timeout=15)
        phases.append(trace("03-oidc-discovery", "GET", r))
        if r.status_code >= 400:
            return _fail("oidc_discovery", r, phases)

        # Phase 04: device/code
        r = s.post(DEVICE_CODE_URL, impersonate=IMP, timeout=15,
                   data={"client_id": CLIENT_ID, "scope": DEFAULT_SCOPE, "referrer": "grok-build"},
                   headers=cli_form_headers(cli_ver, surface="ui"))
        phases.append(trace("04-device-code", "POST", r))
        if r.status_code >= 400:
            return _fail("device_code", r, phases)
        dev = r.json()
        uc, dc, vc = dev["user_code"], dev["device_code"], dev["verification_uri_complete"]
        interval = dev.get("interval", 5)

        # Phase 04a: verify page
        r = s.get(vc, impersonate=IMP, timeout=30, allow_redirects=True,
                  headers=browser_html_headers(ua, "verify_page", uc))
        phases.append(trace("04a-verify-page", "GET", r))
        if r.status_code >= 400:
            return _fail("verify_page", r, phases)

        # Phase 04b: POST verify
        r = s.post(VERIFY_URL, impersonate=IMP, timeout=30, allow_redirects=False,
                   data={"user_code": uc},
                   headers=browser_html_headers(ua, "verify", uc))
        phases.append(trace("04b-device-verify", "POST", r))
        if r.status_code not in (302, 303):
            return _fail("device_verify", r, phases)
        consent_url = r.headers.get("Location", "")

        # Phase 04b-2: GET consent page
        if consent_url:
            r = s.get(consent_url, impersonate=IMP, timeout=30, allow_redirects=True,
                      headers=browser_html_headers(ua, "consent", uc))
            phases.append(trace("04b-consent-page", "GET", r))

        # Phase 04c: POST approve
        r = s.post(APPROVE_URL, impersonate=IMP, timeout=30, allow_redirects=False,
                   data={"user_code": uc, "action": "allow", "principal_type": "User", "principal_id": ""},
                   headers=browser_html_headers(ua, "approve", uc, consent_url))
        phases.append(trace("04c-device-approve", "POST", r))
        if r.status_code not in (302, 303):
            return _fail("device_approve", r, phases)

        # Phase 05: token poll
        for i in range(15):
            time.sleep(interval)
            r = s.post(TOKEN_URL, impersonate=IMP, timeout=15,
                       data={"grant_type": "urn:ietf:params:oauth:grant-type:device_code",
                             "device_code": dc, "client_id": CLIENT_ID},
                       headers=cli_form_headers(cli_ver))
            tok = r.json()
            if "access_token" in tok:
                phases.append(trace("05-token-poll", "POST", r))
                # Phase 05a-05g: enrichment
                identity, bot_flag = _run_enrichment(s, tok["access_token"], cli_ver, agent_id, opts, phases)
                return {"ok": True, "tokens": tok, "identity": identity, "bot_flag": bot_flag, "phases": phases}
            if tok.get("error") in ("expired_token", "access_denied"):
                phases.append(trace("05-token-poll", "POST", r))
                return _fail("token_poll", r, phases)
        return {"ok": False, "error_phase": "token_poll_timeout", "error_status": 0,
                "error_message": "Token poll 超时", "error_url": TOKEN_URL, "phases": phases}

    except Exception as e:
        return {"ok": False, "error_phase": "exception", "error_status": 0,
                "error_message": str(e), "error_url": "", "phases": phases}

def _run_enrichment(s, access_token, cli_ver, agent_id, opts, phases):
    """Phase 05a-05g: user → settings → models → bundle → billing → subscription (fail-open)."""
    identity = {"user_id": "", "email": "", "team_id": ""}
    bot_flag = {"class": "clean", "raw": ""}
    if opts.get("skip_init_user"):
        return identity, bot_flag

    enrich = {"agent_id": agent_id, "include_shell_identifier": True}

    # 05a: /v1/user
    r = s.get(USER_URL, impersonate=IMP, timeout=15, headers=cli_api_headers(access_token, cli_ver))
    phases.append(trace("05a-init-user", "GET", r))
    if r.status_code < 300:
        data = r.json()
        identity["user_id"] = data.get("userId", "")
        identity["email"] = data.get("email", "")
        identity["team_id"] = data.get("teamId", "")
        enrich["user_id"] = identity["user_id"]
        enrich["email"] = identity["email"]

    # 05b: settings
    r = s.get(SETTINGS_URL, impersonate=IMP, timeout=15, headers=cli_enrichment_headers(access_token, cli_ver, enrich))
    phases.append(trace("05b-settings", "GET", r))

    # 05c: models
    r = s.get(MODELS_URL, impersonate=IMP, timeout=15, headers=cli_enrichment_headers(access_token, cli_ver, enrich))
    phases.append(trace("05c-models", "GET", r))

    # 05d: bundle/archive
    r = s.get(BUNDLE_URL, impersonate=IMP, timeout=15, headers=cli_enrichment_headers(access_token, cli_ver, enrich))
    phases.append(trace("05d-bundle", "GET", r))

    # 05e: billing
    r = s.get(BILLING_URL, impersonate=IMP, timeout=15, headers=cli_enrichment_headers(access_token, cli_ver, enrich))
    phases.append(trace("05e-billing", "GET", r))

    # 05g: subscription
    r = s.get(SUBSCRIPTION_URL, impersonate=IMP, timeout=15,
              headers=cli_enrichment_headers(access_token, cli_ver, {"agent_id": agent_id}))
    phases.append(trace("05g-subscription", "GET", r))
    if r.status_code < 300:
        data = r.json()
        if data.get("userId"): identity["user_id"] = data["userId"]
        if data.get("email"): identity["email"] = data["email"]
        if data.get("teamId"): identity["team_id"] = data["teamId"]

    # Bot flag 从 access_token JWT 解析
    claims = _decode_jwt_claims(access_token)
    if claims:
        raw = claims.get("bot_flag_source")
        if raw is None or (isinstance(raw, str) and (raw.strip() == "" or raw.strip().upper() == "NP")):
            bot_flag = {"class": "clean", "raw": str(raw) if raw else ""}
        elif isinstance(raw, (int, float)) and raw == 1:
            bot_flag = {"class": "super_signal", "raw": "1"}
        else:
            bot_flag = {"class": "contaminated", "raw": str(raw)}

    return identity, bot_flag

def _fail(phase, response, phases):
    """构造失败响应。"""
    status = response.status_code if response else 0
    msg = f"sso2oauth[{phase}] 失败 status={status}"
    url = str(response.url) if response and hasattr(response, "url") else ""
    return {"ok": False, "error_phase": phase, "error_status": status,
            "error_message": msg, "error_url": url, "phases": phases}
```

```go
func (a *Adapter) ConvertToBuild(ctx context.Context, credential accountdomain.Credential) (provider.CredentialSeed, error) {
	// 现有：解密 token、AcquireCredential lease（不动）
	token, err := a.cipher.Decrypt(credential.EncryptedAccessToken)
	// ...
	lease, err := a.egress.AcquireCredential(ctx, egressdomain.ScopeWeb, credential)
	// ...

	requestCtx, cancel := context.WithTimeout(ctx, 90*time.Second)
	defer cancel()

	client := a.sso2oauthClient
	a.mu.RUnlock()

		return a.convertViaDaemon(requestCtx, credential, token, lease, client)
	}
	// daemon 不可用 → 回退 Go tls-client + 警告
	a.log().Warn("sso2oauth_daemon_unavailable_fallback",
		"warning", "Go tls-client impersonate 精度不足 (uTLS vs BoringSSL)，CF 拦截概率高")
	// ... 现有 Go flow 代码（不动）
}

func (a *Adapter) convertViaDaemon(ctx context.Context, credential accountdomain.Credential, token string, lease *infraegress.Lease, client *sso2oauth.Client) (provider.CredentialSeed, error) {
	cfg := a.config()
	cliVersion := strings.TrimSpace(cfg.BuildClientVersion)
	if cliVersion == "" {
		cliVersion = xaiauth.DefaultCLIVersion
	}
	browserUA := strings.TrimSpace(lease.UserAgent)
	if browserUA == "" {
		browserUA = xaiauth.DefaultBrowserUA
	}
	req := sso2oauth.ConvertRequest{
		SsoToken:   token,
		ProxyURL:   lease.ProxyURL,
		UserAgent:  browserUA,
		CFCookies:  lease.CFCookies,
		CLIVersion: cliVersion,
		Options: sso2oauth.ConvertOptions{
			SoftPreflight: cfg.ConvertSoftPreflight,
			SkipInitUser:  cfg.SkipConvertInitUser,
			SkipBotReject: cfg.SkipConvertBotReject,
		},
	}
	resp, err := client.Convert(ctx, req)
	if err != nil {
		// daemon 不可达 → 回退 Go flow（不在 convertViaDaemon 内回退，由 ConvertToBuild 处理）
		return provider.CredentialSeed{}, fmt.Errorf("sso2oauth daemon 不可达: %w", err)
	}
	if !resp.OK {
		return provider.CredentialSeed{}, a.classifyDaemonError(resp)
	}
	return a.buildSeedFromDaemon(resp, credential)
}

func (a *Adapter) classifyDaemonError(resp *sso2oauth.ConvertResponse) error {
	phase := resp.ErrorPhase
	msg := resp.ErrorMessage
	status := resp.ErrorStatus
	var wrapped error
	switch {
	case status == http.StatusUnauthorized && strings.Contains(phase, "probe_accounts"):
		wrapped = provider.ErrUnauthorized
	case status == http.StatusTooManyRequests:
		wrapped = conversionHTTPError{status: status}
	case status >= 500 && status < 600:
		wrapped = conversionHTTPError{status: status}
	case status == http.StatusBadRequest && strings.Contains(msg, "access_denied"):
		wrapped = provider.ErrAuthorizationDenied
	case strings.Contains(msg, "bot_flag"):
		wrapped = ErrBuildTokenBotContaminated
	default:
		wrapped = conversionHTTPError{status: status}
	}
	return fmt.Errorf("sso2oauth[%s] %s: %w", phase, msg, wrapped)
}

func (a *Adapter) buildSeedFromDaemon(resp *sso2oauth.ConvertResponse, credential accountdomain.Credential) (provider.CredentialSeed, error) {
	if resp.Tokens == nil {
		return provider.CredentialSeed{}, fmt.Errorf("sso2oauth daemon 返回成功但无 tokens")
	}
	// Go 端独立解析 JWT（不信任 Python 的 identity）
	userID, email, teamID, accessClaims := xaiauth.IdentityFromTokens(resp.Tokens.AccessToken, resp.Tokens.IDToken)
	botClass, botRaw := xaiauth.ClassifyConvertBot(accessClaims)
	cfg := a.config()
	if botClass == xaiauth.ConvertBotContaminated && !cfg.SkipConvertBotReject {
		return provider.CredentialSeed{}, fmt.Errorf("%w: %s", ErrBuildTokenBotContaminated, botRaw)
	}
	// 记录 phase trace 到 Debug 日志
	if a.log().Enabled(context.Background(), slog.LevelDebug) {
		a.log().Debug("sso2oauth_daemon_trace", "phases", resp.Phases)
	}
	name := strings.TrimSpace(credential.Name)
	if name == "" {
		name = "Grok Web account"
	}
	return provider.CredentialSeed{
		Provider: accountdomain.ProviderBuild, AuthType: accountdomain.AuthTypeOAuth,
		Name: firstValue(email, name+" Build", userID, "Grok Build account"), Email: email, UserID: userID, TeamID: teamID,
		SourceKey: "sso-build:" + security.HashToken(resp.Tokens.AccessToken), OIDCClientID: ssoBuildClientID,
		AccessToken: resp.Tokens.AccessToken, RefreshToken: resp.Tokens.RefreshToken, ExpiresAt: time.Now().Add(time.Duration(resp.Tokens.ExpiresIn) * time.Second),
	}, nil
}
```
